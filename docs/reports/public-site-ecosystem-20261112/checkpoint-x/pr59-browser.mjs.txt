async function newPage(browser, { lang = 'es', token = null, viewport = { width: 1280, height: 900 }, api = {} } = {}) {
  const ctx = await browser.newContext({ viewport });
  const log = [];
  await ctx.addInitScript(([lang, token]) => {
    try { localStorage.setItem('noetia_lang', lang); if (token) localStorage.setItem('access_token', token); } catch {}
  }, [lang, token]);
  const page = await ctx.newPage();
  const state = { refreshCalls: 0, meCalls: 0, prefGetCalls: 0 };
  await page.route('**/*', async (route) => {
    const req = route.request();
    const url = new URL(req.url());
    const auth = req.headers()['authorization'] ?? null;
    if (url.origin === BASE && !url.pathname.startsWith('/api/')) return route.continue();
    if (url.host === 'checkout.stripe.test') { log.push({ m: req.method(), p: 'STRIPE_STUB' }); return route.fulfill({ status: 200, contentType: 'text/html', body: '<h1>stub</h1>' }); }
    let path = null;
    if (url.origin === BASE && url.pathname.startsWith('/api/')) path = url.pathname.slice(4);
    else if (url.origin === 'http://localhost:4000') path = url.pathname;
    if (path === null) return route.abort(); // block everything else (fonts are self-hosted)
    const body = req.postData();
    log.push({ m: req.method(), p: path, auth, body });
    const json = (status, obj) => route.fulfill({ status, contentType: 'application/json', body: obj === undefined ? '' : JSON.stringify(obj) });
    const h = api[`${req.method()} ${path}`];
    if (h) return h({ json, auth, body, state, route });
    if (path === '/subscriptions/plans') return json(200, PLANS);
    if (path === '/subscriptions/token-packages') return json(200, PKGS);
    if (path === '/causes' && req.method() === 'GET') return json(200, CAUSES);
    if (path === '/auth/refresh') { state.refreshCalls++; return json(401, { message: 'no session' }); }
    if (!auth) return json(401, { message: 'Unauthorized' });
    if (path === '/subscriptions/me') { state.meCalls++; return json(200, { status: 'none' }); }
    if (path === '/causes/preferences' && req.method() === 'GET') { state.prefGetCalls++; return json(200, undefined); }
    if (path === '/causes/preferences' && req.method() === 'POST') return json(201, JSON.parse(body));
    if (path === '/subscriptions/checkout' || path === '/subscriptions/tokens/purchase') return json(200, { url: STRIPE_STUB });
    return json(404, { message: 'mock: not found' });
  });
  return { ctx, page, log, state };
}
