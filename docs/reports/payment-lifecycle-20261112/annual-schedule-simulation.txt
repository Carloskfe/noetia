const stripeSub: any = { id: 'sub_Y', customer: 'cus_1', status: 'trialing', cancel_at_period_end: false, current_period_end: 0, trial_end: 0, items: { data: [{ price: { id: 'price_y' } }] } };
jest.mock('stripe', () => jest.fn().mockImplementation(() => ({ subscriptions: { retrieve: async () => ({ ...stripeSub }) } })));

const PLAN = { id: 'plan-y', stripePriceId: 'price_y', tokensPerCycle: 1, interval: 'year', maxProfiles: 1 };
const USER = { id: 'u1', stripeCustomerId: 'cus_1' };
let row: any = null;
const grants: Array<{ day: number; reason: string }> = [];
const processed = new Set<string>();
const DAY = 86_400_000;
const T0 = Date.UTC(2027, 0, 1, 12, 0, 0); // trial starts 2027-01-01 12:00 UTC
const dayOf = (ms: number) => Math.round(((ms - T0) / DAY) * 100) / 100;

const subRepo: any = {
  findOneBy: async (w: any) => (row && (!w.userId || w.userId === row.userId) ? { ...row } : null),
  findOne: async ({ where }: any) => (row && (!where.stripeSubscriptionId || where.stripeSubscriptionId === row.stripeSubscriptionId) ? { ...row, plan: row.planId === PLAN.id ? PLAN : null } : null),
  find: async ({ where }: any) => {
    if (!row) return [];
    if (where.stripeSubscriptionId) return where.stripeSubscriptionId === row.stripeSubscriptionId ? [{ ...row }] : [];
    const due = row.status === where.status && row.nextTokenIssuanceAt && row.nextTokenIssuanceAt < where.nextTokenIssuanceAt.value;
    return due ? [{ ...row, plan: PLAN }] : [];
  },
  upsert: async (v: any) => { row = { id: 'row1', linkedUserIds: [], planId: null, nextTokenIssuanceAt: null, ...(row ?? {}), ...v }; },
  update: async (where: any, v: any) => { if (row && (where === row.id || where.id === row.id || where.stripeSubscriptionId === row.stripeSubscriptionId)) row = { ...row, ...v }; },
};
const tokenRepo: any = { create: (v: any) => v, save: async (rows: any[]) => { for (const r of rows) grants.push({ day: dayOf(Date.now()), reason: r.reason }); } };
const userRepo: any = { findOneBy: async () => USER };
const em: any = {
  getRepository: (e: unknown) => (e === Subscription ? subRepo : e === User ? userRepo : e === TokenLedger ? tokenRepo : {}),

const sec = (ms: number) => Math.floor(ms / 1000);

async function runYearPlan(opts: { cancelAtPeriodEndDay?: number; immediateCancelDay?: number; days: number }) {
  // Day 0: trial starts (Checkout); trial ends day 14; annual period then runs 365 days.
  const trialEnd = T0 + 14 * DAY;
  let periodEnd = trialEnd + 365 * DAY;
  at(T0);
  Object.assign(stripeSub, { status: 'trialing', trial_end: sec(trialEnd), current_period_end: sec(trialEnd), cancel_at_period_end: false });
  await ev('customer.subscription.created', { id: 'sub_Y' });
  await ev('invoice.paid', { id: 'in_trial', subscription: 'sub_Y', amount_paid: 0, billing_reason: 'subscription_create' });
  let invoiceN = 0;
  const payInvoice = async (ms: number) => {
    at(ms);
    Object.assign(stripeSub, { status: 'active', trial_end: sec(trialEnd), current_period_end: sec(periodEnd) });
    await ev('invoice.paid', { id: `in_y${++invoiceN}`, subscription: 'sub_Y', amount_paid: 8399, billing_reason: 'subscription_cycle' });
  };
  await payInvoice(trialEnd);
  for (let d = 1; d <= opts.days; d++) {
    const cronAt = Date.UTC(2027, 0, 1 + d, 1, 0, 0); // daily 01:00 UTC
    if (cronAt >= periodEnd && row.status !== 'canceled' && !stripeSub.cancel_at_period_end) {
      periodEnd += 365 * DAY;
      await payInvoice(periodEnd - 365 * DAY); // renewal invoice on the old period end
    }
    if (opts.cancelAtPeriodEndDay !== undefined && d === opts.cancelAtPeriodEndDay) {
      at(cronAt - 3600_000);
      Object.assign(stripeSub, { cancel_at_period_end: true });
      await ev('customer.subscription.updated', { id: 'sub_Y' });
    }
    if (stripeSub.cancel_at_period_end && cronAt >= periodEnd && row.status !== 'canceled') {
      at(periodEnd);
      Object.assign(stripeSub, { status: 'canceled' });
      await ev('customer.subscription.deleted', { id: 'sub_Y' });
    }
    if (opts.immediateCancelDay !== undefined && d === opts.immediateCancelDay) {
      at(cronAt - 3600_000);
      Object.assign(stripeSub, { status: 'canceled' });
      await ev('customer.subscription.deleted', { id: 'sub_Y' });
    }
    at(cronAt);
    await subs.issueMonthlyTokensForAnnualPlans();
  }
  return { periodEndDay: dayOf(trialEnd + 365 * DAY) };
}

function summarize(label: string, periodEndDay: number) {
  const inFirstPaid = grants.filter((g) => g.day >= 14 && g.day < periodEndDay).length;
  console.log(`\n=== ${label} ===\n` +
    grants.map((g) => `day ${g.day.toFixed(2).padStart(7)}  ${g.reason}`).join('\n') +
    `\ntotal grants=${grants.length}; T-1=${grants.filter((g) => g.reason.startsWith('Trial')).length}; ` +
    `paid cycles inside the first paid annual period [day 14, day ${periodEndDay}) = ${inFirstPaid}`);
}
