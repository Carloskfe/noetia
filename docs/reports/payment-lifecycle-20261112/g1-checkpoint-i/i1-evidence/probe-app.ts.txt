// G1-I1 diagnostic upstream (scratch, never committed to src): the SAME @nestjs/throttler 6.5.0 configuration as
// the API (one throttler named 'global', 120/min; the G-1 route overrides it to 30/min) behind a real Traefik v2.11.
// MODE selects Express 'trust proxy': none (= current API), restricted (one private first hop), unrestricted (true).
import 'reflect-metadata';
import { Controller, Get, Module, Req } from '@nestjs/common';
import { APP_GUARD, NestFactory } from '@nestjs/core';
import { SkipThrottle, Throttle, ThrottlerGuard, ThrottlerModule } from '@nestjs/throttler';
import { isIP } from 'net';

const MODE = process.env.MODE ?? 'none';
const PRIVATE = [/^10\./, /^172\.(1[6-9]|2\d|3[01])\./, /^192\.168\./, /^127\./, /^::1$/, /^f[cd]/i, /^::ffff:(10|127|192\.168|172\.(1[6-9]|2\d|3[01]))\./];
const isPrivate = (a: string) => isIP(a) !== 0 && PRIVATE.some((r) => r.test(a));

@Controller()
class ProbeController {
  @Get('whoami') @SkipThrottle({ global: true })
  whoami(@Req() req: any) {
    return { mode: MODE, reqIp: req.ip, reqIps: req.ips, socket: req.socket.remoteAddress, xForwardedFor: req.headers['x-forwarded-for'] ?? null, xRealIp: req.headers['x-real-ip'] ?? null };
  }
  @Get('limited') @Throttle({ global: { limit: 30, ttl: 60_000 } })   // mirrors CheckoutStatusController
  limited() { return { ok: true }; }
  @Get('other')                                                      // any other API route: global 120/min
  other() { return { ok: true }; }
}

@Module({
  imports: [ThrottlerModule.forRoot([{ name: 'global', ttl: 60_000, limit: 120 }])],
  controllers: [ProbeController],
  providers: [{ provide: APP_GUARD, useClass: ThrottlerGuard }],
})
class ProbeModule {}

(async () => {
  const app = await NestFactory.create(ProbeModule, { logger: false });
  const express = app.getHttpAdapter().getInstance();
  if (MODE === 'restricted') express.set('trust proxy', (addr: string, i: number) => i === 0 && isPrivate(addr));
  if (MODE === 'unrestricted') express.set('trust proxy', true);
  await app.listen(Number(process.env.PORT), '0.0.0.0');
  console.log(`probe ${MODE} on ${process.env.PORT}`);
})();
